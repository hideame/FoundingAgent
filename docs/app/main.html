 94@app.get("/", response_class=HTMLResponse)
 95async def read_root(request: Request, session_id: str | None = Cookie(default=None)):
 96    # セッションIDがあればデータをロード
 97    user_tasks = TASKS  # デフォルト
 98    chat_messages_html = ""  # ここで履歴HTML文字列を作る
 99
100    if session_id:
101        data = session_store.load_session(session_id)
102        if data:
103            user_tasks = data.get("tasks", TASKS)
104            history = data.get("chat_history", [])
105
106            # Gemini履歴を復元
107            if history:
108                gemini_service.restore_chat_session(session_id, history)
109
110            # 表示用の履歴HTMLを再構築
111            # ここでは簡易的に、historyの構造から role=user or model を判定してテンプレート適用
112            # 本来は templates/components/message.html を再利用して文字列結合する
113            for msg in history:
114                role = msg["role"]
115                # model -> "text", user -> "text"
116                text_content = ""
117                if "parts" in msg:
118                    for part in msg["parts"]:
119                        if "text" in part:
120                            text_content += part["text"]
121
122                is_user = role == "user"
123
124                msg_html = templates.get_template("components/message.html").render(
125                    {"request": request, "message": text_content, "is_user": is_user}
126                )
127                chat_messages_html += msg_html
128
129    return templates.TemplateResponse(
130        "index.html",
131        {
132            "request": request,
133            "steps": ROADMAP_STEPS,
134            "tasks": user_tasks,
135            "initial_chat_history": chat_messages_html,
136        },
137    )

391@app.get("/plan/download_excel")
392async def download_plan_excel(session_id: str | None = Cookie(default=None)):
393    """
394    保存された創業計画書データをExcelテンプレートに書き込み、ダウンロードさせます。
395    """
396    if not session_id:
397        return Response("Session not found", status_code=400)
398
399    data = session_store.load_session(session_id)
400    if not data or not data.get("plan_text"):
401        return Response("Plan text not found", status_code=404)
402
403    plan_text = data.get("plan_text")
404
405    # テンプレート読み込み
406    template_path = BASE_DIR / "static" / "templates" / "startup_plan_template.xlsx"
407    workbook = openpyxl.load_workbook(template_path)
408    sheet = workbook.active
409
410    # --- plan_textの解析とマッピング ---
411    # テキストからセクションを抽出してセルに埋め込む
412    # 出力フォーマットの揺れに対応できるよう、見出しの前後パターンを緩めに扱う
413
414    heading_order = [
415        ("motivation", "創業の動機"),
416        ("background", "経営者の略歴等"),
417        ("service", "取扱商品・サービス"),
418        ("partners", "取引先・取引関係等"),
419        ("employees", "従業員"),
420        ("loans", "お借入の状況"),
421        ("funds", "必要な資金と調達方法"),
422        ("outlook", "事業の見通し"),
423    ]
424
425    def build_section_pattern(current_label, next_labels):
426        next_part = "|".join(re.escape(label) for label in next_labels)
427        return rf"(?:^|\n)\s*(?:\d+[\.|\s]*)?{re.escape(current_label)}\s*(?:\n|:|：)\s*(.*?)(?=\n\s*(?:\d+[\.|\s]*)?(?:{next_part})\s*(?:\n|:|：)|\Z)"
428
429    sections = {}
430    for idx, (key, label) in enumerate(heading_order):
431        next_labels = [next_label for _, next_label in heading_order[idx + 1 :]]
432        pattern = (
433            build_section_pattern(label, next_labels)
434            if next_labels
435            else build_section_pattern(label, [])
436        )
437        sections[key] = pattern
438
439    def normalize_section_text(content: str) -> str:
440        content = content.strip()
441        content = re.sub(r"[\*#]+", "", content)
442        content = re.sub(r"\n\s*\n+", "\n\n", content)
443        return content.strip()
444
445    def extract_section(pattern, text):
446        match = re.search(pattern, text, re.DOTALL | re.IGNORECASE)
447        if match:
448            return normalize_section_text(match.group(1))
449        return ""
450
451    print(f"[DEBUG] plan_text length: {len(plan_text)}")
452    print(f"[DEBUG] plan_text preview: {plan_text[:500]}")
453
454    # テンプレート内の見出しセルを探して、転記先セルを動的に推定する
455    label_to_key = {
456        "創業の動機": "motivation",
457        "経営者の略歴等": "background",
458        "取扱商品・サービス": "service",
459        "取引先・取引関係等": "partners",
460        "従業員": "employees",
461        "お借入の状況": "loans",
462        "必要な資金と調達方法": "funds",
463        "事業の見通し": "outlook",
464    }
465
466    label_cells = {}
467    for row in sheet.iter_rows():
468        for cell in row:
469            if isinstance(cell.value, str):
470                value = cell.value.strip()
471                for label, key in label_to_key.items():
472                    if label in value:
473                        label_cells[key] = cell
474
475    def choose_target_cell(label_cell):
476        label_row = label_cell.row
477        label_col = label_cell.column
478        candidates = []
479        for merged_range in sheet.merged_cells.ranges:
480            min_col, min_row, max_col, max_row = range_boundaries(str(merged_range))
481            if min_row >= label_row + 1 and min_row <= label_row + 12:
482                width = max_col - min_col + 1
483                height = max_row - min_row + 1
484                if width >= 6:
485                    if min_col <= label_col <= max_col or min_col > label_col:
486                        candidates.append(
487                            (min_row, -(width * height), min_col, str(merged_range))
488                        )
489        if candidates:
490            candidates.sort()
491            selected = candidates[0][3]
492            min_col, min_row, _, _ = range_boundaries(selected)
493            return f"{get_column_letter(min_col)}{min_row}"
494        return f"{get_column_letter(label_col + 1)}{label_row}"
495
496    mapping = {key: None for key in label_to_key.values()}
497    for key, cell in label_cells.items():
498        mapping[key] = choose_target_cell(cell)
499
500    # フォールバック用の静的マッピング（テンプレート検出できない場合）
501    fallback_mapping = {
502        "motivation": "A9",
503        "background": "A16",
504        "service": "A27",
505        "partners": "M27",
506        "employees": "T21",
507        "loans": "M36",
508        "funds": "A45",
509        "outlook": "M45",
510    }
511    for key, addr in fallback_mapping.items():
512        if not mapping.get(key):
513            mapping[key] = addr
514
515    # セル結合されていることが多いので、左上のセルに値をセットする
516    extracted_any = False
517    for key, pattern in sections.items():
518        content = extract_section(pattern, plan_text)
519        print(f"[DEBUG] Extracted {key}: {content[:100] if content else '(empty)'}...")
520        cell_addr = mapping.get(key)
521        if cell_addr and content:
522            extracted_any = True
523            # 結合セルの場合でも安全に値を設定
524            try:
525                cell = sheet[cell_addr]
526                # MergedCellの場合、unmergeしてから書き込む
527                if isinstance(cell, openpyxl.cell.cell.MergedCell):
528                    # 結合範囲を探して解除
529                    for merged_range in list(sheet.merged_cells.ranges):
530                        if cell.coordinate in merged_range:
531                            sheet.unmerge_cells(str(merged_range))
532                            break
533                # 値を設定
534                sheet[cell_addr].value = content
535                print(f"[DEBUG] Successfully wrote to {cell_addr}")
536            except Exception as e:
537                print(f"Warning: Could not write to cell {cell_addr}: {e}")
538                continue
539
540    if not extracted_any and plan_text.strip():
541        fallback_cell = mapping.get("motivation")
542        if fallback_cell:
543            try:
544                sheet[fallback_cell].value = normalize_section_text(plan_text)
545                print(
546                    "[DEBUG] No sections matched. Wrote full plan_text to motivation cell as fallback."
547                )
548            except Exception as e:
549                print(f"Warning: Could not write fallback plan_text: {e}")
550
551    # メモリ上のバイナリとして保存 (Excel)
552    excel_output = BytesIO()
553    workbook.save(excel_output)
554    excel_output.seek(0)

457        "創業の動機": "motivation",

459        "取扱商品・サービス": "service",
460        "取引先・取引関係等": "partners",

465    }

467    label_cells = {}
468    for row in sheet.iter_rows():
469        for cell in row:
470            if isinstance(cell.value, str):
471                value = cell.value.strip()
472                for label, key in label_to_key.items():
473                    if label in value:
474                        label_cells[key] = cell
475
476    def choose_target_cell(label_cell):
477        label_row = label_cell.row
478        label_col = label_cell.column
479        candidates = []
480        for merged_range in sheet.merged_cells.ranges:
481            min_col, min_row, max_col, max_row = range_boundaries(str(merged_range))
482            if min_row >= label_row + 1 and min_row <= label_row + 12:
483                width = max_col - min_col + 1
484                height = max_row - min_row + 1
485                if width >= 6:
486                    if min_col <= label_col <= max_col or min_col > label_col:
487                        candidates.append(
488                            (min_row, -(width * height), min_col, str(merged_range))
489                        )
490        if candidates:
491            candidates.sort()
492            selected = candidates[0][3]
493            min_col, min_row, _, _ = range_boundaries(selected)
494            return f"{get_column_letter(min_col)}{min_row}"
495        return f"{get_column_letter(label_col + 1)}{label_row}"
496
497    mapping = {key: None for key in label_to_key.values()}
498    for key, cell in label_cells.items():
499        mapping[key] = choose_target_cell(cell)
500
501    # フォールバック用の静的マッピング（テンプレート検出できない場合）
502    fallback_mapping = {
503        "motivation": "A9",
504        "background": "A16",
505        "service": "A27",
506        "partners": "M27",
507        "employees": "T21",
508        "loans": "M36",
509        "funds": "A45",
510        "outlook": "M45",
511    }

514            mapping[key] = addr
515
516    # セル結合されていることが多いので、左上のセルに値をセットする
517    extracted_any = False
518    for key, pattern in sections.items():
519        content = extract_section(pattern, plan_text)
520        print(f"[DEBUG] Extracted {key}: {content[:100] if content else '(empty)'}...")
521        cell_addr = mapping.get(key)
522        if cell_addr and content:
523            extracted_any = True
524            # 結合セルの場合でも安全に値を設定
525            try:
526                cell = sheet[cell_addr]
527                # MergedCellの場合、unmergeしてから書き込む
528                if isinstance(cell, openpyxl.cell.cell.MergedCell):
529                    # 結合範囲を探して解除
530                    for merged_range in list(sheet.merged_cells.ranges):
531                        if cell.coordinate in merged_range:
532                            sheet.unmerge_cells(str(merged_range))
533                            break
534                # 値を設定
535                sheet[cell_addr].value = content
536                print(f"[DEBUG] Successfully wrote to {cell_addr}")
537            except Exception as e:
538                print(f"Warning: Could not write to cell {cell_addr}: {e}")
539                continue
540
541    if not extracted_any and plan_text.strip():
542        fallback_cell = mapping.get("motivation")
543        if fallback_cell:
544            try:
545                sheet[fallback_cell].value = normalize_section_text(plan_text)
546                print(
547                    "[DEBUG] No sections matched. Wrote full plan_text to motivation cell as fallback."
548                )
549            except Exception as e:
550                print(f"Warning: Could not write fallback plan_text: {e}")
551
552    # メモリ上のバイナリとして保存 (Excel)
553    excel_output = BytesIO()
554    workbook.save(excel_output)
555    excel_output.seek(0)
556
557    # 業種判定とPDF選択
558    examples_dir = BASE_DIR / "static" / "templates" / "examples"
559    pdf_filename = None
560
561    # Determine keywords based on plan_text
562    # Simple keyword matching to guess industry
563    if re.search(r"飲食|居酒屋|カフェ|レストラン|食堂|ランチ|ディナー|料理", plan_text):
564        pdf_filename = "restaurant_example.pdf"
565    elif re.search(r"美容|サロン|ヘア|ネイル|エステ|カット|パーマ", plan_text):
566        pdf_filename = "beauty_example.pdf"
567    elif re.search(r"自動車|中古車|車両|整備|板金|ガソリン", plan_text):
568        pdf_filename = "car_sales_example.pdf"
569    elif re.search(

573        pdf_filename = "apparel_example.pdf"
574    elif re.search(

576    ):

578    elif re.search(

580    ):
581        pdf_filename = "cram_school_example.pdf"
582    elif re.search(

586    elif re.search(

594        pdf_filename = "software_example.pdf"

597    zip_output = BytesIO()
